cmake_minimum_required(VERSION 3.16)
project(factory_board_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT LVGL_SOURCE_DIR)
  set(LVGL_SOURCE_DIR "${REPO_ROOT}/.build/factory-components/lvgl")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
add_executable(rotation_test rotation_test.cpp)
target_include_directories(rotation_test PRIVATE "${REPO_ROOT}/firmware/factory_badge/main")
target_link_libraries(rotation_test PRIVATE lvgl::lvgl)
add_executable(touch_test touch_test.cpp)
target_include_directories(touch_test PRIVATE "${REPO_ROOT}/firmware/factory_badge/main")
target_link_libraries(touch_test PRIVATE lvgl::lvgl)
add_executable(rtc_test rtc_test.cpp "${REPO_ROOT}/firmware/factory_badge/main/vendor/rx8130/rx8130.cpp")
target_include_directories(rtc_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${REPO_ROOT}/firmware/factory_badge/main")
set(M5GFX_SOURCE_DIR "${REPO_ROOT}/.build/factory-components/M5GFX" CACHE PATH "Pinned M5GFX source")
add_executable(flush_test flush_test.cpp
  "${M5GFX_SOURCE_DIR}/src/lgfx/v1/panel/Panel_FrameBufferBase.cpp"
  "${M5GFX_SOURCE_DIR}/src/lgfx/v1/panel/Panel_Device.cpp"
  "${M5GFX_SOURCE_DIR}/src/lgfx/v1/misc/pixelcopy.cpp"
  "${M5GFX_SOURCE_DIR}/src/lgfx/v1/misc/common_function.cpp")
target_include_directories(flush_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${REPO_ROOT}/firmware/factory_badge/main" "${M5GFX_SOURCE_DIR}/src")
target_compile_options(flush_test PRIVATE -UNDEBUG)
foreach(target rotation_test rtc_test touch_test)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endforeach()
enable_testing()
add_test(NAME factory_rotation COMMAND rotation_test)
add_test(NAME factory_rtc COMMAND rtc_test)
add_test(NAME factory_touch_provenance COMMAND touch_test)
add_test(NAME factory_flush COMMAND flush_test)
