cmake_minimum_required(VERSION 3.16)
project(badge_ui_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(FACTORY_COMPONENTS "${REPO_ROOT}/.build/factory-components" CACHE PATH "Pinned factory component sources")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(MOONCAKE_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
option(FACTORY_UI_SANITIZERS "Run address/undefined behavior sanitizers" ON)
if(FACTORY_UI_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory("${FACTORY_COMPONENTS}/lvgl" lvgl)
add_subdirectory("${FACTORY_COMPONENTS}/mooncake" mooncake)
set(UI_ROOT "${REPO_ROOT}/firmware/factory_badge/main")
file(GLOB UI_SOURCES CONFIGURE_DEPENDS "${UI_ROOT}/ui/*.cpp")
add_executable(factory_ui_test main.cpp "${UI_ROOT}/badge_ui.cpp" ${UI_SOURCES})
target_include_directories(factory_ui_test PRIVATE "${UI_ROOT}"
    "${FACTORY_COMPONENTS}/smooth_ui_toolkit/src"
    "${FACTORY_COMPONENTS}/smooth_ui_toolkit/src/lvgl")
target_link_libraries(factory_ui_test PRIVATE lvgl mooncake)
enable_testing()
add_test(NAME factory_ui COMMAND factory_ui_test)
