cmake_minimum_required(VERSION 3.16)
project(badge_intro_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(FACTORY_COMPONENTS "${REPO_ROOT}/.build/factory-components" CACHE PATH "Pinned factory component sources")
set(LV_BUILD_CONF_PATH "${REPO_ROOT}/tests/factory-ui/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory("${FACTORY_COMPONENTS}/lvgl" lvgl)
add_executable(factory_intro_test main.cpp "${REPO_ROOT}/firmware/factory_badge/main/ui/intro_loop.c")
target_include_directories(factory_intro_test PRIVATE "${REPO_ROOT}/firmware/factory_badge/main/ui")
target_link_libraries(factory_intro_test PRIVATE lvgl)
enable_testing()
add_test(NAME factory_intro COMMAND factory_intro_test)
set_tests_properties(factory_intro PROPERTIES ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1")
